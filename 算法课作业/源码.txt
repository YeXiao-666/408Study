 //回溯法实现  
class BacktrackSolution {
private:
	int maxValue; // 记录最大价值
	vector<int> remainSum; // 后缀价值和（基于按照单位价值排序后的顺序）

public:
	int cutStone(int L, vector<int>& lengths, vector<int>& values) {
		maxValue = 0;
		int n = lengths.size();

		vector<pair<double, int>> orders; // (单位价值, 原始索引)
		for (int i = 0; i < n; ++i) 
		{
			orders.emplace_back(static_cast<double>(values[i]) / lengths[i], i);// 按单位价值降序排序（关键剪枝优化）
		}
		sort(orders.begin(), orders.end(), [](const auto& a, const auto& b)
		{
				return a.first > b.first; // 降序排序
		});


		vector<int> sortedLengths(n), sortedValues(n);
		for (int i = 0; i < n; ++i) 
		{
			sortedLengths[i] = lengths[orders[i].second];
			sortedValues[i] = values[orders[i].second];//按排序后的顺序重新排列 lengths 和 values
		}

		// 计算排序后的后缀价值和
		remainSum.resize(n);
		remainSum[n - 1] = sortedValues[n - 1];
		for (int i = n - 2; i >= 0; --i) 
		{
			remainSum[i] = remainSum[i + 1] + sortedValues[i];
		}

		//调用回溯函数
		backtrack(0, 0, 0, L, sortedLengths, sortedValues);
		return maxValue;
	}

private:
	void backtrack(int start, int currentLength, int currentValue,
		int L, const vector<int>& lengths, const vector<int>& values) {
		// 剪枝1：剩余订单的总价值无法超过当前最大值
		if (start < lengths.size() && currentValue + remainSum[start] <= maxValue) {
			return;
		}

		// 更新最大值
		if (currentValue > maxValue) {
			maxValue = currentValue;
		}

		// 遍历订单
		for (int i = start; i < lengths.size(); ++i) {
			// 剪枝2：长度超过限制
			if (currentLength + lengths[i] > L) {
				continue;
			}

			// 递归探索
			backtrack(i + 1, currentLength + lengths[i], currentValue + values[i],
				L, lengths, values);
		}
	}
};
// 2. 分支限界法实现
class BranchAndBoundSolution {
private:
	// 定义优先队列中的节点结构
	struct Node {
		int currentLevel;				// 当前处理到的石材层级
		int currentLength;				// 已使用的总长度
		int currentValue;				// 已获得的总价值
		double currentBound;			// 当前节点的价值上界

		Node(int l, int len, int val, double b)
			: currentLevel(l), currentLength(len), currentValue(val), currentBound(b) {
		}

		// 重载<运算符用于优先队列（最大堆）
		bool operator<(const Node& other) const
		{
			return currentBound < other.currentBound;//优先队列默认是最大堆
		}
	};

public:
	int cutStone(int L, vector<int>& lengths, vector<int>& values)
	{
		int n = lengths.size();
		vector<pair<double, int>> orders;// 存储（单位价值，索引）的向量	

		for (int i = 0; i < n; i++)
		{
			orders.emplace_back(values[i] / (double)lengths[i], i); //存入（单位价值，索引）
		}

		// 按单位价值降序排序
		sort(orders.begin(), orders.end(),
			[](const pair<double, int>& a, const pair<double, int>& b)
			{
				return a.first > b.first;//降序排列
			}); // lambda表达式

		int maxValue = 0;// 记录最大价值
		priority_queue<Node> heap; // 优先队列（最大堆）
		// 初始化根节点：层级0，长度0，价值0，计算初始上界
		heap.emplace(0, 0, 0, bound(0, 0, 0, L, lengths, values, orders));

		while (!heap.empty())
		{
			Node node = heap.top();// 取出上界最大的节点
			heap.pop();

			// 剪枝：如果当前上界小于已知最大价值，终止搜索
			if (node.currentBound <= maxValue) continue;

			// 到达叶子节点时更新最大值
			if (node.currentLevel == n)
			{
				if (node.currentValue > maxValue)
				{
					maxValue = node.currentValue;
				}
				continue;
			}
			// 选择当前石材的分支（如果能放入）
			if (node.currentLength + lengths[node.currentLevel] <= L)
			{
				int newValue = node.currentValue + values[node.currentLevel];
				if (newValue > maxValue)
				{
					maxValue = newValue;
				}
				// 计算新节点的上界
				double newBound = bound(node.currentLevel + 1, node.currentLength + lengths[node.currentLevel], newValue, L, lengths, values, orders);
				// 只有上界可能超过当前最大值时才入队
				if (newBound > maxValue)
				{
					heap.emplace(node.currentLevel + 1, node.currentLength + lengths[node.currentLevel], newValue, newBound);
				}
			}

			// 不选择当前石材的分支
			double boundValue = bound(node.currentLevel + 1, node.currentLength, node.currentValue, L, lengths, values, orders);
			if (boundValue > maxValue)
			{
				heap.emplace(node.currentLevel + 1, node.currentLength, node.currentValue, boundValue);
			}
		}

		return maxValue;//返回结果
	}

private:
	// 计算上界函数（贪心思想）
	double bound(int level, int currentLength, int currentValue, int L, vector<int>& lengths, vector<int>& values, vector<pair<double, int>>& orders)
	{
		double boundValue = currentValue;// 初始化上界为当前价值
		int remaining = L - currentLength;// 剩余可用长度
		for (auto it = orders.begin(); it != orders.end(); ++it) //// 遍历排序后的石材列表
		{
			if (it->second < level) continue; // 跳过已处理的层级（根据索引判断）
			if (lengths[it->second] <= remaining)
			{
				boundValue += values[it->second];  // 累加价值
				remaining -= lengths[it->second]; // 减少剩余长度
			}
			else
			{
				break;// 剩余空间不足，终止计算
			}
		}

		return boundValue;//返回估算的上界值	
	}
};

// 3. 动态规划实现
class DpSolution {
public:
	int cutStone(int L, vector<int>& lengths, vector<int>& values) {
		int n = lengths.size();

		// 二维DP表：dp[i][j]表示前i个石材在容量j时的最大价值
		vector<vector<int>> dp(n + 1, vector<int>(L + 1, 0));
		// 构建DP表
		for (int i = 1; i <= n; i++) // 遍历每个石材
		{
			for (int j = 1; j <= L; j++) // 遍历每个容量
			{
				if (lengths[i - 1] <= j) // 如果当前石材可以放入
				{
					dp[i][j] = max(dp[i - 1][j],// 不选当前石材的情况
						values[i - 1] + dp[i - 1][j - lengths[i - 1]]);// 选择当前石材的情况
				}

				else 
				{
					dp[i][j] = dp[i - 1][j];// 无法放入，继承之前的结果
				}
			}
		}

		return dp[n][L];  // 返回最终结果
	}
};

// 4. 优化动态规划实现
class OptimizedDpSolution {
public:
	int cutStone(int L, vector<int>& lengths, vector<int>& values)
	{
		int n = lengths.size();
		vector<int> dp(L + 1, 0);// 一维DP数组
		for (int i = 0; i < n; i++) // 遍历每个石材
		{
			for (int j = L; j >= lengths[i]; j--) // 逆序遍历容量（避免覆盖之前的状态）
			{
				dp[j] = max(dp[j], values[i] + dp[j - lengths[i]]);// 状态转移方程：max(不选当前，选当前+剩余空间的最优解)
			}
		}

		return dp[L];// 返回最终结果
	}

};
void performanceComparison() {
	vector<int> sizes = { 5, 10, 15, 20, 25, 35 };
	random_device rd;
	mt19937 gen(rd());
	uniform_int_distribution<> len_dist(1, 20);
	uniform_int_distribution<> val_dist(1, 40);

	BacktrackSolution backtrack;
	BranchAndBoundSolution bb;
	DpSolution dp;
	OptimizedDpSolution optDp;

	cout << "\n性能比较 (单位: 毫秒):\n";
	cout << "大小\t回溯法\t分支限界\t动态规划\t优化DP\n";

	for (int n : sizes) {
		int L = n * 5;
		vector<int> lengths(n);
		vector<int> values(n);

		for (int i = 0; i < n; i++) {
			lengths[i] = len_dist(gen);
			values[i] = val_dist(gen);
		}

		auto start = chrono::high_resolution_clock::now();
		backtrack.cutStone(L, lengths, values);
		auto end = chrono::high_resolution_clock::now();
		double backtrack_time = chrono::duration_cast<chrono::microseconds>(end - start).count() / 1000.0;

		start = chrono::high_resolution_clock::now();
		bb.cutStone(L, lengths, values);
		end = chrono::high_resolution_clock::now();
		double bb_time = chrono::duration_cast<chrono::microseconds>(end - start).count() / 1000.0;

		start = chrono::high_resolution_clock::now();
		dp.cutStone(L, lengths, values);
		end = chrono::high_resolution_clock::now();
		double dp_time = chrono::duration_cast<chrono::microseconds>(end - start).count() / 1000.0;

		start = chrono::high_resolution_clock::now();
		optDp.cutStone(L, lengths, values);
		end = chrono::high_resolution_clock::now();
		double optDp_time = chrono::duration_cast<chrono::microseconds>(end - start).count() / 1000.0;

		cout << n << "\t" << backtrack_time << "\t" << bb_time << "\t\t"
			<< dp_time << "\t\t" << optDp_time << endl;
	}
}
